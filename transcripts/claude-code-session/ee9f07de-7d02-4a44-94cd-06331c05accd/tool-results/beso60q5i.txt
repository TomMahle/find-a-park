   264	/**
   265	 * Leaflet popups don't manage focus. This moves focus into a popup when it
   266	 * opens (with ParkPopup's help on the first open), puts it back on whatever
   267	 * opened it (a marker or a "Show on map" button) when it closes, and makes
   268	 * Escape close the popup, so keyboard users never lose their place.
   269	 */
   270	function PopupFocusManager() {
   271	  const map = useMap();
   272	
   273	  useEffect(() => {
   274	    let opener: HTMLElement | null = null;
   275	
   276	    const onOpen = (event: L.PopupEvent) => {
   277	      const active = document.activeElement;
   278	      // On phones the "Show on map" button is hidden as the map appears, so
   279	      // focus has already fallen back to <body>; treat that as no opener.
   280	      opener =
   281	        active instanceof HTMLElement && active !== document.body
   282	          ? active
   283	          : null;
   284	      // react-leaflet keeps popup content mounted after the first open, so on
   285	      // later opens the link already exists and can take focus now. (On the
   286	      // first open it doesn't exist yet; ParkPopup focuses it on mount.)
   287	      event.popup
   288	        .getElement()
   289	        ?.querySelector<HTMLElement>("[data-popup-primary]")
   290	        ?.focus({ preventScroll: true });
   291	    };
   292	
   293	    const onClose = (event: L.PopupEvent) => {
   294	      const container = event.popup.getElement();
   295	      const active = document.activeElement;
   296	      const focusWasInPopup =
   297	        active === null ||
   298	        active === document.body ||
   299	        (container?.contains(active) ?? false);
   300	      const returnTo = opener;
   301	      opener = null;
   302	      if (!focusWasInPopup) return;
   303	
   304	      // Prefer the element that opened the popup. If it is gone or hidden
   305	      // (the mobile list is hidden while the map shows), use the marker.
   306	      if (returnTo?.isConnected && isVisible(returnTo)) {
   307	        returnTo.focus({ preventScroll: true });
   308	        return;
   309	      }
   310	      const parkId = container
   311	        ?.querySelector<HTMLElement>("[data-popup-primary]")
   312	        ?.getAttribute("data-popup-primary");
   313	      if (parkId) {
   314	        map
   315	          .getContainer()
   316	          .querySelector<HTMLElement>(
   317	            `.park-marker[data-park-id="${CSS.escape(parkId)}"]`,
   318	          )
   319	          ?.focus({ preventScroll: true });
   320	      }
   321	    };
   322	
   323	    // Leaflet only handles Escape while the map container itself has focus,
   324	    // not while focus is inside a popup or on a marker.
   325	    const onKeyDown = (event: KeyboardEvent) => {
   326	      if (event.key === "Escape") map.closePopup();
   327	    };
   328	    const container = map.getContainer();
   329	
   330	    map.on("popupopen", onOpen);
   331	    map.on("popupclose", onClose);
   332	    container.addEventListener("keydown", onKeyDown);
   333	    return () => {
   334	      map.off("popupopen", onOpen);
   335	      map.off("popupclose", onClose);
   336	      container.removeEventListener("keydown", onKeyDown);
   337	    };
   338	  }, [map]);
   339	
   340	  return null;
   341	}

    21	/**
    22	 * A drop-in replacement for react-leaflet's `MapContainer` that survives
    23	 * Next.js navigation.
    24	 *
    25	 * With `cacheComponents` on, Next.js keeps a page you navigate away from
    26	 * mounted inside a hidden React `<Activity>`: its effects are cleaned up, its
    27	 * DOM is kept, and the effects run again when you come back. react-leaflet's
    28	 * `MapContainer` destroys the Leaflet map in that cleanup but only ever
    29	 * creates it once, so on return every marker tries to attach to a destroyed
    30	 * map and throws.
    31	 *
    32	 * This version creates the map once and only destroys it when its element
    33	 * has really left the document. While hidden, the DOM stays, so the map
    34	 * survives, layers re-attach to it, and the user's pan and zoom are kept.
    35	 */
    36	export function StableMapContainer({
    37	  options,
    38	  bounds,
    39	  boundsOptions,
    40	  className,
    41	  children,
    42	}: StableMapContainerProps) {
    43	  const nodeRef = useRef<HTMLDivElement>(null);
    44	  const mapRef = useRef<L.Map | null>(null);
    45	  const [context, setContext] = useState<LeafletContextInterface | null>(null);
    46	  const initial = useRef({ options, bounds, boundsOptions });
    47	
    48	  useEffect(() => {
    49	    const node = nodeRef.current;
    50	    if (!node) return;
    51	
    52	    if (!mapRef.current) {
    53	      const map = L.map(node, initial.current.options);
    54	      map.fitBounds(initial.current.bounds, initial.current.boundsOptions);
    55	      mapRef.current = map;
    56	      // Children (tiles, markers) can only render once the map exists.
    57	      setContext(createLeafletContext(map));
    58	    }
    59	
    60	    return () => {
    61	      // React removes an unmounted component's DOM before running its effect
    62	      // cleanups, so a still-connected node means the page was only hidden
    63	      // (or this is React Strict Mode's simulated remount in development).
    64	      if (!node.isConnected) {
    65	        mapRef.current?.remove();
    66	        mapRef.current = null;
    67	      }
    68	    };
    69	  }, []);
    70	
    71	  return (
    72	    <div ref={nodeRef} className={className}>
    73	      {context && <LeafletContext value={context}>{children}</LeafletContext>}
